#include "klee/klee.h"
#include "klee/file_api.h"
#include <fcntl.h>
int main(void) {
    int fd = __file_open("A");
    klee_assert(fd >= 0);

    klee_assert(__file_flags(fd) == O_RDWR);      /* opened O_RDWR */
    __file_set_flags(fd, O_RDONLY);
    klee_assert(__file_flags(fd) == O_RDONLY);    /* flags translation */

    klee_assert(__file_offset(fd) == 0);
    __file_set_offset(fd, 5);
    klee_assert(__file_offset(fd) == 5);          /* setter reaches f->off */

    __file_set_size(fd, 99);
    klee_assert(__file_size(fd) == 99);           /* getter reads dfile->size */
    return 0;
}
EOF
